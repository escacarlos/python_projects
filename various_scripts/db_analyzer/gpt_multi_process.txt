from excel_parser_testqc import analyze_extract_testqc
from excel_parser_autocontrole import analyze_extract_autocontrole
import pandas as pd
import os


def prepare_resultat_and_csv(
    testqc_path,
    autocontrole_path,
    liberation_path,
    is_old_version,
    output_filename="merged_results.csv",
    max_time_diff_hours=2
):
    # --------------------------------------------------
    # 1. Load data
    # --------------------------------------------------

    records_testqc = analyze_extract_testqc(testqc_path)
    records_autocontrole = analyze_extract_autocontrole(
        autocontrole_path, liberation_path, is_old_version
    )

    cols_rqc = [
        'combined_datetime_rqc', 'of_number_rqc', 'product_code_rqc',
        'lot_number_rqc', 'description_rqc', 'conditionnement_rqc',
        'largeur_rqc', 'longueur_rqc',
        'thick_T1_rqc', 'thick_T2_rqc', 'thick_T3_rqc', 'thick_rqc'
    ]

    cols_a = [
        'combined_datetime_a', 'of_number_a', 'product_code_a',
        'lot_number_a', 'description_a',
        't_min_a', 't_max_a', 'test_qc_a',
        'largeur_a', 'longueur_a', 'pente_a',
        'thick_T1_a', 'thick_T2_a', 'thick_T3_a', 'thick_a'
    ]

    df_rqc = pd.DataFrame(records_testqc, columns=cols_rqc)
    df_a = pd.DataFrame(records_autocontrole, columns=cols_a)

    # --------------------------------------------------
    # 2. Normalize datatypes
    # --------------------------------------------------

    df_rqc['combined_datetime_rqc'] = pd.to_datetime(df_rqc['combined_datetime_rqc'])
    df_a['combined_datetime_a'] = pd.to_datetime(df_a['combined_datetime_a'])

    # Normalize OF numbers (CRITICAL)
    df_rqc['of_number_rqc'] = df_rqc['of_number_rqc'].astype(str).str.strip()
    df_a['of_number_a'] = df_a['of_number_a'].astype(str).str.strip()

    # --------------------------------------------------
    # 3. Find nearest datetime per OF number
    # --------------------------------------------------

    grouped_autocontrole = df_a.groupby('of_number_a')
    max_delta = pd.Timedelta(hours=max_time_diff_hours)

    # Prepare empty matched DF aligned with RQC
    df_a_matched = pd.DataFrame(index=df_rqc.index, columns=df_a.columns)

    for rqc_idx, row_rqc in df_rqc.iterrows():

        of_number = row_rqc['of_number_rqc']
        target_dt = row_rqc['combined_datetime_rqc']

        # OF not present at all
        if of_number not in grouped_autocontrole.groups:
            continue

        df_group = df_a.loc[grouped_autocontrole.groups[of_number]].copy()

        # Compute absolute time difference
        df_group['time_diff'] = (df_group['combined_datetime_a'] - target_dt).abs()

        # Enforce max allowed delta
        df_group = df_group[df_group['time_diff'] <= max_delta]

        if df_group.empty:
            continue

        # Select closest match
        best_idx = df_group['time_diff'].idxmin()
        df_a_matched.loc[rqc_idx] = df_a.loc[best_idx]

    # --------------------------------------------------
    # 4. Final merge
    # --------------------------------------------------

    df_final = pd.concat(
        [
            df_rqc,
            df_a_matched.drop(columns=['of_number_a', 'combined_datetime_a'])
        ],
        axis=1
    )

    # --------------------------------------------------
    # 5. Output CSV
    # --------------------------------------------------

    folder_path = os.path.dirname(testqc_path)
    output_dir = os.path.dirname(folder_path.rstrip("/\\"))
    output_csv = os.path.join(output_dir, output_filename)

    df_final.to_csv(output_csv, index=False)
    print(f"Merged records data saved as: {output_csv}")

    return df_final

In multiprocessing.Pool, the key point is how arguments are passed to the worker function.
There are three common, clean patterns depending on what kind of extra arguments you have.

1️⃣ Extra arguments are the same for all files (most common)

Use functools.partial to “freeze” the extra arguments.

from multiprocessing import Pool, cpu_count
from functools import partial

def analyze_extract_autocontrole(folder_path, arg1, arg2):

    print(f"Found {len(excel_files)} files")

    records = []

    worker = partial(process_one_file, arg1=arg1, arg2=arg2)

    with Pool(processes=cpu_count()) as pool:
        for result in pool.imap_unordered(worker, excel_files):
            records.extend(result)

    print(f"{len(records)} lines written")
    return records


And your worker looks like:

def process_one_file(excel_file, arg1, arg2):
    ...


✅ Best when the extra arguments are constants (config, flags, paths, etc.)

2️⃣ Different arguments per file

Use starmap or imap_unordered with tuples.

Using starmap (simplest)
def analyze_extract_autocontrole(folder_path):
    excel_files = list_excel_files(folder_path)

    tasks = [(file, "A", 42) for file in excel_files]

    with Pool(processes=cpu_count()) as pool:
        for result in pool.starmap(process_one_file, tasks):
            records.extend(result)

    return records


Worker:

    ...

        for result in pool.imap_unordered(lambda args: process_one_file(*args), tasks):
            records.extend(result)

    return records

Good question — what you’re seeing is very common when converting I/O-heavy Excel code to multiprocessing.
Short answer: there is no infinite loop, but there are multiple serious inefficiencies that can easily make the multiprocessing version slower than the synchronous one.

I’ll go through this in a structured way and point out the real performance killers, then give concrete fixes.

✅ No infinite loop

First, the good news:

All loops (os.walk, for i in range(13, 76)) are bounded

No while loops

No recursive calls

No shared state mutation that could re-trigger work

➡️ There is no logical infinite loop.

❌ Why it’s slower than the synchronous version
1️⃣ You are pickling HUGE objects for every task (BIGGEST ISSUE)

This line is the main culprit:

worker = partial(analyze_extract_one_file, liberation_data, database_data)

Why this is bad

multiprocessing copies arguments to every process

liberation_data and especially database_data are likely large

They are pickled and sent to each worker

This happens once per worker process, but still costs a lot

If database_data is large → this alone can make multiprocessing slower.

🔥 Symptom

High CPU usage at startup

Long delay before files are processed

More processes = slower startup

2️⃣ openpyxl is I/O-bound, not CPU-bound

You are doing:

openpyxl.load_workbook(..., read_only=True)


Excel parsing is:

Disk I/O heavy

Python-object heavy

Not pure CPU math

Multiprocessing only shines when:

Tasks are CPU-heavy

Minimal shared data

Minimal disk contention

Here:

All processes read Excel files

All processes hit the disk

OS disk cache contention occurs

➡️ Multiple processes fight for disk access.

3️⃣ SQLite / DB lookup in multiprocessing

You pass database_data to each process and call:

lookup_from_db_load(database_data, ...)


If database_data:

Is a DB connection → ❌ unsafe across processes

Is a big in-memory structure → ❌ heavy pickling

Either way, DB access in multiprocessing is often slower than single-process batching.

4️⃣ Too many processes
Pool(processes=cpu_count())


This is usually wrong for I/O-bound workloads.

For Excel + disk:

cpu_count() is often too high

Causes context switching + I/O contention

5️⃣ Printing from worker processes
print(f"{file}: OF number {of_number} not found...")


Printing from multiple processes:

Is synchronized internally

Slows everything down

Can look like the program is “stuck”

⚠️ Smaller inefficiencies (not fatal)

Repeated sheet[f"A{i}"] lookups (slow but unavoidable)

Repeated datetime.combine

Repeated exception handling inside tight loops

These are minor compared to #1–#4.
